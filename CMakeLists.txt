cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
project(fxp_math VERSION 1.0.0 LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(FXP_TOP_LEVEL ON)
else()
    set(FXP_TOP_LEVEL OFF)
endif()

option(FXP_BUILD_TESTS "Build the GoogleTest suite" ${FXP_TOP_LEVEL})
option(FXP_FETCH_GOOGLETEST "Download pinned GoogleTest if no package is installed" ON)
option(FXP_BUILD_EXAMPLES "Build examples" ${FXP_TOP_LEVEL})
option(FXP_BUILD_BENCHMARKS "Build fixed/float/double comparison benchmarks" OFF)
option(FXP_PORTABLE_ONLY "Use only standard C++ two-word wide arithmetic" OFF)
option(FXP_DISABLE_SSE2 "Disable SSE2 batch operations" OFF)
option(FXP_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(fxp_math INTERFACE)
add_library(fxp::fxp ALIAS fxp_math)
set_target_properties(fxp_math PROPERTIES EXPORT_NAME fxp)
target_compile_features(fxp_math INTERFACE cxx_std_17)
target_include_directories(fxp_math INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(FXP_PORTABLE_ONLY)
    target_compile_definitions(fxp_math INTERFACE FXP_PORTABLE_ONLY=1)
endif()
if(FXP_DISABLE_SSE2)
    target_compile_definitions(fxp_math INTERFACE FXP_DISABLE_SSE2=1)
endif()

function(fxp_project_options target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
        if(FXP_ENABLE_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

if(FXP_BUILD_TESTS)
    enable_testing()
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
        find_package(GTest QUIET)
    endif()
    if(NOT GTest_FOUND)
        if(NOT FXP_FETCH_GOOGLETEST)
            message(FATAL_ERROR "GoogleTest was not found. Install GTest or enable FXP_FETCH_GOOGLETEST.")
        endif()
        include(FetchContent)
        set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(googletest
            URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz
            URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(fxp_tests tests/test_fixed.cpp tests/test_math.cpp tests/test_ieee.cpp tests/test_reference.cpp tests/test_batch.cpp tests/test_tangent_regression.cpp tests/test_geometry.cpp tests/test_geometry_tolerances.cpp)
    target_link_libraries(fxp_tests PRIVATE fxp::fxp GTest::gtest_main)
    target_compile_definitions(fxp_tests PRIVATE FXP_REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    fxp_project_options(fxp_tests)
    # Every public header must compile independently, without transitive umbrella includes.
    foreach(header fixed math ieee754 batch vector geometry geometry_misc fxp)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.cpp" "#include <fxp/${header}.hpp>\n")
        add_library(fxp_header_${header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.cpp")
        target_link_libraries(fxp_header_${header} PRIVATE fxp::fxp)
        fxp_project_options(fxp_header_${header})
    endforeach()
    add_executable(fxp_reference_writer tools/reference_writer.cpp)
    target_link_libraries(fxp_reference_writer PRIVATE fxp::fxp)
    # The canonical outputs always come from the standard C++ scalar backend,
    # independently of the configuration used to test optimized implementations.
    target_compile_definitions(fxp_reference_writer PRIVATE FXP_PORTABLE_ONLY=1 FXP_DISABLE_SSE2=1)
    fxp_project_options(fxp_reference_writer)
    add_executable(fxp_geometry_reference_writer tools/geometry_reference_writer.cpp)
    target_link_libraries(fxp_geometry_reference_writer PRIVATE fxp::fxp)
    target_compile_definitions(fxp_geometry_reference_writer PRIVATE FXP_PORTABLE_ONLY=1 FXP_DISABLE_SSE2=1)
    fxp_project_options(fxp_geometry_reference_writer)
    include(GoogleTest)
    gtest_discover_tests(fxp_tests DISCOVERY_TIMEOUT 60)
    # Build the valid control normally too, so header regressions fail the build.
    # Intentionally use only public consumer requirements here: on MSVC this
    # exercises the default parser without our private /permissive- option.
    add_library(fxp_compile_accept OBJECT tests/compile_fail/accept.cpp)
    target_link_libraries(fxp_compile_accept PRIVATE fxp::fxp)
    if(MSVC)
        # Check the same constructor constraints with the conforming parser.
        add_library(fxp_compile_accept_conforming OBJECT tests/compile_fail/accept.cpp)
        target_link_libraries(fxp_compile_accept_conforming PRIVATE fxp::fxp)
        fxp_project_options(fxp_compile_accept_conforming)
    endif()
    foreach(case float_construct double_assign float_add float_cast)
        # These intentionally invalid targets are never part of a normal build.
        # Reuse the actual toolchain instead of maintaining nested CMake caches.
        add_library(fxp_reject_${case} OBJECT EXCLUDE_FROM_ALL tests/compile_fail/${case}.cpp)
        target_link_libraries(fxp_reject_${case} PRIVATE fxp::fxp)
        add_test(NAME compile_reject_${case} COMMAND ${CMAKE_COMMAND}
            "-DBINARY_DIR=${CMAKE_BINARY_DIR}" "-DTARGET=fxp_reject_${case}" "-DCONFIG=$<CONFIG>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/RejectFloat.cmake")
        # Multiple independent build-tool processes must not mutate the same
        # build tree simultaneously when CTest runs with --parallel.
        set_tests_properties(compile_reject_${case} PROPERTIES RESOURCE_LOCK fxp_compile_reject_build)
    endforeach()
endif()
if(FXP_BUILD_EXAMPLES)
    add_executable(fxp_example examples/game.cpp)
    target_link_libraries(fxp_example PRIVATE fxp::fxp)
    fxp_project_options(fxp_example)
endif()
if(FXP_BUILD_BENCHMARKS)
    add_executable(fxp_benchmark benchmarks/benchmark.cpp)
    target_link_libraries(fxp_benchmark PRIVATE fxp::fxp)
    set(FXP_BENCHMARK_CONFIGURATION_FLAGS
        "$<$<CONFIG:Debug>:${CMAKE_CXX_FLAGS_DEBUG}>$<$<CONFIG:Release>:${CMAKE_CXX_FLAGS_RELEASE}>$<$<CONFIG:RelWithDebInfo>:${CMAKE_CXX_FLAGS_RELWITHDEBINFO}>$<$<CONFIG:MinSizeRel>:${CMAKE_CXX_FLAGS_MINSIZEREL}>")
    configure_file(benchmarks/benchmark_config.hpp.in
        "${CMAKE_CURRENT_BINARY_DIR}/benchmark_config.hpp.in" @ONLY)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/benchmark/$<CONFIG>/benchmark_config.hpp"
        INPUT "${CMAKE_CURRENT_BINARY_DIR}/benchmark_config.hpp.in")
    target_include_directories(fxp_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/benchmark/$<CONFIG>")
    target_compile_definitions(fxp_benchmark PRIVATE FXP_BENCHMARK_CONFIG="$<CONFIG>")
    fxp_project_options(fxp_benchmark)
endif()

    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fxp)

    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fxpConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/fxpConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fxp)
